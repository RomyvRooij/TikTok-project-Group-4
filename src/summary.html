3. Overall performance

overview <- views %>%
  summarise(
    videos = n(),
    creators = n_distinct(creator_id),
    avg_length = mean(video_length_sec, na.rm = TRUE),
    avg_impressions = mean(impressions_n, na.rm = TRUE),
    avg_watched = mean(watched_n, na.rm = TRUE),
    avg_watch_rate = mean(watch_rate, na.rm = TRUE),
    avg_watch_share = mean(watched_n / impressions_n, na.rm = TRUE),
    avg_watch_seconds = mean(
      video_length_sec * watched_n / impressions_n,
      na.rm = TRUE
    )
  )
